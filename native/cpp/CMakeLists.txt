cmake_minimum_required(VERSION 3.18)
project(novelreader_native_text LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ICU REQUIRED COMPONENTS uc i18n)
add_library(novel_text STATIC src/novel_text.cpp)
target_include_directories(novel_text PUBLIC include)
target_link_libraries(novel_text PUBLIC ICU::uc ICU::i18n)

option(NOVEL_ENABLE_CT2 "Build the CTranslate2 + SentencePiece inference worker" OFF)
if(NOVEL_ENABLE_CT2)
  find_package(ctranslate2 CONFIG REQUIRED)
  find_package(sentencepiece CONFIG REQUIRED)
  find_package(nlohmann_json CONFIG REQUIRED)
  add_executable(novelreader-ct2-worker src/ct2_worker.cpp)
  target_link_libraries(novelreader-ct2-worker PRIVATE ctranslate2 sentencepiece nlohmann_json::nlohmann_json)
endif()

include(CTest)
if(BUILD_TESTING)
  add_executable(test_novel_text tests/test_novel_text.cpp)
  target_link_libraries(test_novel_text PRIVATE novel_text)
  add_test(NAME novel_text_unicode COMMAND test_novel_text)
endif()
