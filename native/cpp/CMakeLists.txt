cmake_minimum_required(VERSION 3.18)
project(novelreader_native_text LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ICU REQUIRED COMPONENTS uc i18n)
add_library(novel_text STATIC src/novel_text.cpp)
target_include_directories(novel_text PUBLIC include)
target_link_libraries(novel_text PUBLIC ICU::uc ICU::i18n)
include(CTest)
if(BUILD_TESTING)
  add_executable(test_novel_text tests/test_novel_text.cpp)
  target_link_libraries(test_novel_text PRIVATE novel_text)
  add_test(NAME novel_text_unicode COMMAND test_novel_text)
endif()
